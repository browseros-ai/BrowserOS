await page.goto(args.base + '/sdk-fill.html');
await page.getByRef(await refFor(page, "Month field")).fill("2026-10");
const state = await page.evaluate(id => ({ value: document.getElementById(id).value, events: window.nativeEvents.filter(event => event.id === id) }), "month");
equal(state.value, "2026-10", 'fill reads back the native value');
equal(state.events, [{ id: "month", kind: 'input', value: "2026-10" }, { id: "month", kind: 'change', value: "2026-10" }], 'fill emits input and change with the stored value');
return state;
