await page.goto(args.base + '/links.html');
const full = await page.ariaSnapshot({ mode: 'full' });
const interactive = await page.ariaSnapshot({ text: false });
assert(!/- paragraph/.test(interactive), 'interactive snapshot drops paragraph rows', interactive);
includes(interactive, '[ref=', 'interactive snapshot retains actionable refs');
assert(interactive.length < full.length, 'interactive snapshot is smaller than full', { full, interactive });
return { full, interactive };
