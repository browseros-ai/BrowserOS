// locator.ref pins the same node for every action and read. Unlike an AI snapshot,
// it can also name a covered field, so that the original refusal can be measured.
async function g1cField(label, scope = page) {
  const ref = await scope.getByLabel(label, { exact: true }).ref({ timeout: 5000 });
  return { label, ref, target: page.getByRef(ref) };
}

/** Read the SDK's YAML field line without imposing desktop's JSON quoting. */
function g1cShown(snapshot, label) {
  const name = JSON.stringify(label);
  const line = snapshot.split('\n').find(row => row.includes(name));
  assert(line !== undefined, 'snapshot contains the filled field', { label, snapshot });
  const afterName = line.slice(line.indexOf(name) + name.length).trim();
  const tail = afterName.replace(/^(?:\[[^\]]+\]\s*)*/, '');
  assert(tail === '' || tail === ':' || tail.startsWith(': '),
    'field line has a scalar readback', { label, line });
  const scalar = tail === '' || tail === ':' ? '' : tail.slice(2);
  const value = scalar.startsWith('"') ? JSON.parse(scalar)
    : scalar.startsWith("'") ? scalar.slice(1, -1).replace(/''/g, "'") : scalar;
  return { label, line, value, snapshotRef: line.match(/\[ref=([^\]]+)\]/)?.[1] ?? null };
}

await page.goto(args.base + '/form.html');
await page.evaluate(url => new Promise(resolve => {
  const frame = document.createElement('iframe');
  frame.onload = () => resolve(true);
  frame.src = url;
  document.body.prepend(frame);
}), args.base2 + '/fill.html');
const field = await g1cField('Text field', page.frameLocator('iframe'));
await field.target.fill('frame text', { timeout: 8000 });
const filledValue = await field.target.inputValue();
equal(filledValue, 'frame text', 'the referenced child field receives the text');
const filledSnapshot = await page.ariaSnapshot({ mode: 'ai' });
const filled = g1cShown(filledSnapshot, field.label);
equal(filled.value, 'frame text', 'the parent snapshot independently exposes the child value');
await field.target.fill('', { timeout: 8000 });
const clearedValue = await field.target.inputValue();
equal(clearedValue, '', 'the referenced child field is empty');
const clearedSnapshot = await page.ariaSnapshot({ mode: 'ai' });
const cleared = g1cShown(clearedSnapshot, field.label);
equal(cleared.value, '', 'the parent snapshot independently exposes the cleared child');
const parentValue = await page.evaluate(() => document.getElementById('name').value);
equal(parentValue, '', 'the parent field is untouched');
return { ref: field.ref, filledValue, filled, clearedValue, cleared, parentValue };
