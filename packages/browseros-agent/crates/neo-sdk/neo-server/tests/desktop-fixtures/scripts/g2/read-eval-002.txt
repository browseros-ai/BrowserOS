await page.goto(args.base + '/links.html');
const text = await page.locator('body').innerText();
const matches = text.split('\n').filter(line => /downloadable report/i.test(line));
assert(matches.some(line => line.includes('downloadable report')), 'content search finds the fixture paragraph', {matches});
assert(!/\[ref=e\d+\]/.test(matches.join('\n')), 'content search contains no accessibility refs', {matches});
return {matches, count:matches.length};
