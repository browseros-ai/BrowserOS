const refIn = (snapshot, label) => {
  const line = snapshot.split('\n').find(line => line.includes(label));
  const ref = line && /\[ref=([^\]]+)\]/.exec(line)?.[1];
  assert(ref, `snapshot must give ${label} its own ref`, { line, snapshot });
  return ref;
};
const selected = (snapshot, labels) => Object.fromEntries(labels.map(label => [label, refIn(snapshot, label)]));
const labelsIn = (snapshot, labels) => {
  for (const label of labels) includes(snapshot, label, `snapshot includes ${label}`);
};

const frameLabels = ["Parent action ready", "Frame A action ready", "Frame A cursor ready", "Grandchild action ready", "Grandchild cursor ready", "Frame B action ready", "Frame B cursor ready"];
const ordered = snapshot => {
  let prior = -1;
  for (const label of frameLabels) {
    const index = snapshot.indexOf(label, prior + 1);
    assert(index !== -1, `frame text preserves DOM stitch order at ${label}`, { snapshot });
    prior = index;
  }
};
const frameSnapshot = async (wanted = frameLabels) => {
  let snapshot = '';
  await poll(async () => {
    const ready = await page.evaluate(() => window.snapshotFrameFixture.state().ready);
    if (!ready) return false;
    snapshot = await page.ariaSnapshot();
    return wanted.every(label => snapshot.includes(label));
  }, 'all frame labels appear in the snapshot', 30000);
  return snapshot;
};
const mixedUrl = new URL(args.base + '/snapshot-frame-tree.html');
const childOrigin = new URL(args.base2);
childOrigin.hostname = mixedUrl.hostname === 'localhost' ? '127.0.0.1' : 'localhost';
mixedUrl.searchParams.set('childOrigin', childOrigin.origin);

await page.goto(mixedUrl.toString());
const initial = await frameSnapshot();
const refs = selected(initial, frameLabels);
await page.getByRef(refs['Frame A action ready']).click();
const afterA = await frameSnapshot(['Frame A action clicked', 'Frame B action ready', 'Grandchild action ready']);
labelsIn(afterA, ['Parent action ready', 'Frame B cursor ready']);
await page.getByRef(refs['Frame B action ready']).click();
const afterB = await frameSnapshot(['Frame A action clicked', 'Frame B action clicked', 'Grandchild action ready']);
labelsIn(afterB, ['Parent action ready', 'Frame A cursor ready']);
await page.getByRef(refs['Grandchild action ready']).click();
const final = await frameSnapshot(['Frame A action clicked', 'Frame B action clicked', 'Grandchild action clicked']);
labelsIn(final, ['Parent action ready', 'Frame A cursor ready', 'Frame B cursor ready']);
return { initialRefs: refs, final };
