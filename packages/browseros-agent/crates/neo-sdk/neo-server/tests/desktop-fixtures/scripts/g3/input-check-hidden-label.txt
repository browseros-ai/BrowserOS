await page.goto(args.base + '/sdk-input.html');

// A selector on the input would bypass the original label-ref contract.
const ref = await refFor(page, 'Styled checkbox label');
const targetTag = await page.getByRef(ref).evaluate(node => node.tagName);
equal(targetTag, 'LABEL', 'the ref names the visible label itself');
for (const [method, checked] of [['check', true], ['uncheck', false]]) {
  await page.getByRef(ref)[method]();
  equal(await page.evaluate(() => document.getElementById('label-control').checked), checked, `${method} follows the label to its hidden control`);
}
return { ref, checked: await page.evaluate(() => document.getElementById('label-control').checked) };
