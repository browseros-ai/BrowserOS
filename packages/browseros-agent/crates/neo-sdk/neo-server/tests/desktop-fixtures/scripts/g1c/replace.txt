// Playwright's label text includes a textarea's initial child text. Use the
// fixture's exact id for that node, then retain the same node-bound ref checks.
async function g1cField(label, scope = page) {
  const locator = label === 'Multiline' ? scope.locator('#multiline')
    : scope.getByLabel(label, { exact: true });
  const ref = await locator.ref({ timeout: 5000 });
  return { label, ref, target: page.getByRef(ref) };
}

await page.goto(args.base + '/fill.html');
const measured = [];
for (const [label, id, requested] of [
  ['Text field', 'text', 'new text 😀'],
  ['Multiline', 'multiline', 'line one\nline two'],
  ['Rich editor', 'rich', 'new rich\ntext'],
  ['Email field', 'email', 'new@example.com'],
  ['Number field', 'number', '123'],
  ['Password field', 'password', 'new password'],
]) {
  const field = await g1cField(label);
  await field.target.fill(requested, { timeout: 8000 });
  const actual = await page.evaluate(id => {
    const element = document.getElementById(id);
    return { value: element.value ?? element.innerText,
      events: window.fillEvents.filter(event => event.id === id) };
  }, id);
  equal(actual.value, requested, label + ' has the exact replacement');
  assert(actual.events.some(event => event.trusted === true),
    label + ' received a trusted input event', actual);
  measured.push({ label, ref: field.ref, requested, ...actual });
}
return { measured };
