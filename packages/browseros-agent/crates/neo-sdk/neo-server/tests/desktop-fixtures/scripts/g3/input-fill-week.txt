await page.goto(args.base + '/sdk-fill.html');
await page.getByRef(await refFor(page, "Week field")).fill("2026-W44");
const state = await page.evaluate(id => ({ value: document.getElementById(id).value, events: window.nativeEvents.filter(event => event.id === id) }), "week");
equal(state.value, "2026-W44", 'fill reads back the native value');
equal(state.events, [{ id: "week", kind: 'input', value: "2026-W44" }, { id: "week", kind: 'change', value: "2026-W44" }], 'fill emits input and change with the stored value');
return state;
