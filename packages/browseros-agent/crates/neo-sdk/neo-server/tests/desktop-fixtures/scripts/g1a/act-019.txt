// A ref is opaque. SDK child-frame refs include a document prefix, unlike desktop eN refs.
const g1aRef = (snapshot, needle) => {
  const row = snapshot.split('\n').find(line =>
    line.replace(/\s+"/g, '"').includes(needle) && /\[ref=([^\]\s]+)\]/.test(line));
  assert(row, 'snapshot has a ref for ' + needle, snapshot);
  return row.match(/\[ref=([^\]\s]+)\]/)[1];
};
await page.goto(args.base + '/scroll.html');
const snapshot = await page.ariaSnapshot();
const before = await page.locator('#drag-order').textContent();
await page.getByRef(g1aRef(snapshot, 'item-a')).dragTo(page.getByRef(g1aRef(snapshot, 'item-c')));
const after = await page.locator('#drag-order').textContent();
assert(before.trim() !== after.trim(), 'drag reorders the list', { before, after });
return { before, after };
