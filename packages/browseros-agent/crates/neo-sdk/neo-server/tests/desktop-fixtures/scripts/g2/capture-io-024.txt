/** Native chooser handles retain the exact input; event registration precedes the activating input. */
function file(name, text) { return {name, mimeType:'text/plain', buffer:Buffer.from(text)}; }
async function choose(p, action) {
  const waiting = p.waitForEvent('filechooser', {timeout:5000});
  await action();
  const chooser = await waiting;
  assert(chooser.page() === p, 'chooser belongs to the initiating page');
  return chooser;
}
