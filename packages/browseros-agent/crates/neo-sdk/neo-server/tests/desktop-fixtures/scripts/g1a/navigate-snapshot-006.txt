// A ref is opaque. SDK child-frame refs include a document prefix, unlike desktop eN refs.
const g1aRef = (snapshot, needle) => {
  const row = snapshot.split('\n').find(line =>
    line.replace(/\s+"/g, '"').includes(needle) && /\[ref=([^\]\s]+)\]/.test(line));
  assert(row, 'snapshot has a ref for ' + needle, snapshot);
  return row.match(/\[ref=([^\]\s]+)\]/)[1];
};
await page.goto(args.base + '/form.html');
const snapshot = await page.ariaSnapshot();
const refs = [...snapshot.matchAll(/\[ref=([^\]\s]+)\]/g)].map(match => match[1]);
assert(refs.length > 0, 'the default snapshot mints refs', snapshot);
const controls = {};
for (const needle of ['Submit', 'Name', 'Color', 'Subscribe']) controls[needle] = g1aRef(snapshot, needle);
return { refs, controls, snapshot };
