// locator.ref pins the same node for every action and read. Unlike an AI snapshot,
// it can also name a covered field, so that the original refusal can be measured.
async function g1cField(label, scope = page) {
  const ref = await scope.getByLabel(label, { exact: true }).ref({ timeout: 5000 });
  return { label, ref, target: page.getByRef(ref) };
}

await page.goto(args.base + '/fill.html');
const field = await g1cField('Shadow field');
await field.target.fill('shadow text', { timeout: 8000 });
const value = await page.evaluate(() => document.getElementById('shadow-host')
  .shadowRoot.querySelector('input').value);
equal(value, 'shadow text', 'the shadow field receives the text');
return { value, ref: field.ref };
