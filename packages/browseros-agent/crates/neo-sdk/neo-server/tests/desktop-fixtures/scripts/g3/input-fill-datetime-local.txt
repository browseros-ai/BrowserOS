await page.goto(args.base + '/sdk-fill.html');
await page.getByRef(await refFor(page, "Datetime field")).fill("2026-10-29T14:30");
const state = await page.evaluate(id => ({ value: document.getElementById(id).value, events: window.nativeEvents.filter(event => event.id === id) }), "datetime-local");
equal(state.value, "2026-10-29T14:30", 'fill reads back the native value');
equal(state.events, [{ id: "datetime-local", kind: 'input', value: "2026-10-29T14:30" }, { id: "datetime-local", kind: 'change', value: "2026-10-29T14:30" }], 'fill emits input and change with the stored value');
return state;
