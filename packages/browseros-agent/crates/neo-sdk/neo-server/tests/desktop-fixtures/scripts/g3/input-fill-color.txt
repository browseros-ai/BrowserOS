await page.goto(args.base + '/sdk-fill.html');
await page.getByRef(await refFor(page, "Color field")).fill("#ff8800");
const state = await page.evaluate(id => ({ value: document.getElementById(id).value, events: window.nativeEvents.filter(event => event.id === id) }), "color");
equal(state.value, "#ff8800", 'fill reads back the native value');
equal(state.events, [{ id: "color", kind: 'input', value: "#ff8800" }, { id: "color", kind: 'change', value: "#ff8800" }], 'fill emits input and change with the stored value');
return state;
