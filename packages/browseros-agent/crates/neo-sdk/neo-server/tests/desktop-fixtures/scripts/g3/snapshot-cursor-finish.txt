const refIn = (snapshot, label) => {
  const line = snapshot.split('\n').find(line => line.includes(label));
  const ref = line && /\[ref=([^\]]+)\]/.exec(line)?.[1];
  assert(ref, `snapshot must give ${label} its own ref`, { line, snapshot });
  return ref;
};
const selected = (snapshot, labels) => Object.fromEntries(labels.map(label => [label, refIn(snapshot, label)]));
const labelsIn = (snapshot, labels) => {
  for (const label of labels) includes(snapshot, label, `snapshot includes ${label}`);
};

const cursorStable = ["Cursor candidate 00", "Cursor candidate 32", "Cursor candidate 63"];
const cursorReasons = ["Pointer-only target", "Assigned onclick target", "Editable target ready", "Tabindex target"];
const untouched = async label => {
  const state = await page.evaluate(() => window.snapshotCursorFixture.state());
  equal(state.mutations, 0, label + ' makes no DOM mutation');
  equal(state.sentinelPresent, true, label + ' preserves the page-owned look-alike marker');
  return state;
};

const captures = args.prior[1];
assert(Array.isArray(captures) && captures.length === 4, 'the host delivers four overlapping MCP capture results', captures);
const expected = captures[0].refs;
for (const [index, capture] of captures.entries()) equal(capture.refs, expected, `overlapping capture ${index + 1} preserves selected refs`);
const afterOverlap = await untouched('overlapping cursor snapshots');
const snapshot = await page.ariaSnapshot();
equal(selected(snapshot, cursorStable), expected, 'a final snapshot preserves selected refs after overlap');
const afterFinal = await untouched('the final snapshot');
return { snapshot, refs: expected, afterOverlap, afterFinal };
