// A ref is opaque. SDK child-frame refs include a document prefix, unlike desktop eN refs.
const g1aRef = (snapshot, needle) => {
  const row = snapshot.split('\n').find(line =>
    line.replace(/\s+"/g, '"').includes(needle) && /\[ref=([^\]\s]+)\]/.test(line));
  assert(row, 'snapshot has a ref for ' + needle, snapshot);
  return row.match(/\[ref=([^\]\s]+)\]/)[1];
};
await page.goto(args.base + '/overlay.html');
const snapshot = await page.ariaSnapshot();
const error = await rejected(() => page.getByRef(g1aRef(snapshot, 'Covered button')).dblclick({ timeout: 1500 }), /overlay|cover|intercept|obscur/i);
const result = await page.locator('#overlay-result').textContent();
includes(result, 'untouched', 'a covered double click does not activate the target');
return { error, result, snapshot };
